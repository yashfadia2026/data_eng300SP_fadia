def impute_carrier_name_airline_id(df):
    df = df.copy()
    mask_l4 = (df['CARRIER'] == 'L4') & df['CARRIER_NAME'].isnull()
    df.loc[mask_l4, 'CARRIER_NAME'] = 'Lynx Aviation d/b/a Frontier Airlines'
    df.loc[mask_l4, 'AIRLINE_ID'] = 21217.0

    mask_oh = (df['CARRIER'] == 'OH') & df['CARRIER_NAME'].isnull()
    df.loc[mask_oh, 'CARRIER_NAME'] = 'Comair Inc.'
    df.loc[mask_oh, 'AIRLINE_ID'] = 20417.0

    return df

These insights highlight that the model of the aircraft is fully observed and has strong predictive power over the missing value for a majority of seat count and capacity cases. Thus, this provides strong support that the data is missing at random (MAR). This eliminates both mean and median imputation as they fail to account for the observed relationship between model and the target variables. This leads to predictive mean matching (PMM) becoming advantageous as it leverages these exact relationships (KNN was preferred but it required significant computation time).

import miceforest as mf

def impute_seats_capacity_pmm(df, num_datasets = 1, num_iterations = 5, mean_match_candidates = 5, random_state = 42):
    df = df.copy()
    df['MODEL_CODE'] = df['MODEL'].astype('category').cat.codes
    # Ensuring a numeric matrix for mice forest
    numeric_columns_in_set = ['MANUFACTURE_YEAR', 'MODEL_CODE', 'NUMBER_OF_SEATS', 'CAPACITY_IN_POUNDS']

    kernel_pmm = mf.ImputationKernel(
        data = df[numeric_columns_in_set], 
        num_datasets = num_datasets,
        mean_match_candidates = mean_match_candidates, 
        random_state = random_state, 
        save_all_iterations_data = False
    )
    kernel_pmm.mice(num_iterations)
    post_imputation = kernel_pmm.complete_data(dataset = 0)
    df['NUMBER_OF_SEATS'] = post_imputation['NUMBER_OF_SEATS']
    df['CAPACITY_IN_POUNDS'] = post_imputation['CAPACITY_IN_POUNDS']
    df.drop(columns = 'MODEL_CODE', inplace = True)
    return df

inventory = impute_seats_capacity_pmm(inventory)

def standardize_manufactuer(df):
    df = df.copy()
    df['MANUFACTURER'] = df['MANUFACTURER'].str.upper()
    df['MANUFACTURER'] = df['MANUFACTURER'].str.strip()
    df['MATCHING_KEY'] = df['MANUFACTURER'].str.replace(r'[\s\.\-\(\)/]', '', regex = True)

    # Now, the matching key will be used to map the rows to the actual name of the manufacturer through a simple string lookup.
    dict_manufacturer = {
        'BOEING': 'BOEING',
        'AIRBUS': 'AIRBUS',
        'MCDONNELL|MCDO': 'MCDONNELL DOUGLAS',
        'BOMBARDIER': 'BOMBARDIER',
        'EMBRAER': 'EMBRAER',
        'CANADAIR': 'CANADAIR',
        'CESSNA': 'CESSNA',
        'DOUGLAS': 'DOUGLAS',
        'DEHAV': 'DE HAVILLAND',
        'SAAB': 'SAAB',
        'GULFSTREAM': 'GULFSTREAM',
        'LOCKHEED': 'LOCKHEED',
        'DASSAULT': 'DASSAULT',
        'DORNIER': 'DORNIER',
        'ATR': 'ATR',
        'FOKKER': 'FOKKER', 
        'RAYTHEON|RAETHEON': 'RAYTHEON',
        'BAE': 'BAE',
        'PIPER|PIPISTREL': 'PIPER', 
        'CURTISS': 'CURTISS',
        'GE': 'GENERAL ELECTRIC',
    }

    set_conditions = [df['MATCHING_KEY'].str.contains(i, na = False) for i in dict_manufacturer.keys()]
    options = list(dict_manufacturer.values())
    df['MANUFACTURER'] = np.select(set_conditions, options, default = df['MANUFACTURER'])
    df.drop(columns = 'MATCHING_KEY', inplace = True)
    return df

inventory = standardize_manufactuer(inventory)

def standardize_model(df):
    df = df.copy()
    df['MODEL'] = df['MODEL'].str.strip()
    df['MODEL'] = df['MODEL'].str.upper()
    df['MODEL'] = df['MODEL'].replace('0', np.nan)
    df['MODEL'] = df['MODEL'].str.replace(r'\s+', '', regex = True) # needed to remove internal spacing for the model name
    df['MODEL'] = df['MODEL'].str.replace(r'-+', '-', regex = True) # removes multiple dashes in the name of the model
    df['MODEL'] = df['MODEL'].str.replace(r'^(7[34567][0-9])', r'B\1', regex = True)
    df['MODEL'] = df['MODEL'].str.replace(r'^(3[012][0-9])', r'A\1', regex = True) # adding prefixes to models
    df['MODEL'] = df['MODEL'].str.replace(r'^([A-Z])-', r'\1', regex = True) # removes the dashes between prefix and model name
    return df

inventory = standardize_model(inventory)

def boxcox_transformation(df):
    df = df.copy()
    # To apply the boxcox transformation, the data needs to be wholly positive. Thus, 1 is added to the columns to ensure all values are positive and non-zero.
    df['NUMBER_OF_SEATS_BOXCOX'] = df['NUMBER_OF_SEATS'] + 1
    df['CAPACITY_IN_POUNDS_BOXCOX'] = df['CAPACITY_IN_POUNDS'] + 1
    df['NUMBER_OF_SEATS_BOXCOX'] = scipy.stats.boxcox(df['NUMBER_OF_SEATS_BOXCOX'])[0]
    df['CAPACITY_IN_POUNDS_BOXCOX'] = scipy.stats.boxcox(df['CAPACITY_IN_POUNDS_BOXCOX'])[0]
    return df 

inventory = boxcox_transformation(inventory)
